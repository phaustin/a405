def find_thetaes(temp,press,rv):
    """ input temp in K, press in Pa, rv in kg/kg
        output thetae in K
    """
    thetaes = temp*(c.p0/press)**(c.Rd/c.cpd)*np.exp(c.lv0*rv/(c.cpd*temp))
    return thetaes
